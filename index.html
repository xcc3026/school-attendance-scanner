// 默认 07:30 } if (timeVal instanceof Date) { const formatted = Utilities.formatDate(timeVal, timeZone, "HH:mm"); const parts = formatted.split(":"); return parseInt(parts[0], 10) * 60 + parseInt(parts[1], 10);
